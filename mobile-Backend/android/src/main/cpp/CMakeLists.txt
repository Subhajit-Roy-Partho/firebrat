cmake_minimum_required(VERSION 3.22.1)
project(firebrat_llama LANGUAGES C CXX)
include(FetchContent)
set(CMAKE_BUILD_TYPE Release CACHE STRING "Optimized mobile inference" FORCE)
set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
find_program(FIREBRAT_GLSLC glslc HINTS /opt/homebrew/bin NO_CMAKE_FIND_ROOT_PATH)
if(FIREBRAT_GLSLC)
  FetchContent_Declare(vulkan_headers
    URL https://codeload.github.com/KhronosGroup/Vulkan-Headers/tar.gz/refs/tags/v1.4.365
    URL_HASH SHA256=ed832b3292cce324ebba6522fd2d3004bf1b797fa49bfbcad0b302ff23d56abc)
  FetchContent_MakeAvailable(vulkan_headers)
  set(Vulkan_INCLUDE_DIR "${vulkan_headers_SOURCE_DIR}/include" CACHE PATH "" FORCE)
  set(Vulkan_GLSLC_EXECUTABLE "${FIREBRAT_GLSLC}" CACHE FILEPATH "" FORCE)
  set(GGML_VULKAN ON CACHE BOOL "" FORCE)
else()
  message(STATUS "glslc not found: building CPU fallback (install shaderc/glslc for Vulkan)")
  set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
endif()
# This revision is the exact native ABI identified by llama_cpp_dart 0.2.2.
FetchContent_Declare(llama_native
  URL https://codeload.github.com/ggml-org/llama.cpp/tar.gz/4ffc47cb2001e7d523f9ff525335bbe34b1a2858
  URL_HASH SHA256=e3bbf8c33761329d225e747fc51f2e9d9291f25be9cbbe545ccba3799795889b)
FetchContent_MakeAvailable(llama_native)
set(LLAMA_INSTALL_VERSION 0.0.0)
add_subdirectory(${llama_native_SOURCE_DIR}/tools/mtmd ${llama_native_BINARY_DIR}/mtmd)
# ELF page alignment is required on current 16 KB-page Android devices.
foreach(native_target mtmd llama ggml ggml-base ggml-cpu ggml-vulkan)
  if(TARGET ${native_target})
    target_link_options(${native_target} PRIVATE "-Wl,-z,max-page-size=16384")
  endif()
endforeach()
